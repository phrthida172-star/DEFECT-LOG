บันทึกข้อมูลการแจ้งปัญหาการผลิต (NC Report)

กรอกรายละเอียดปัญหาที่พบ ข้อมูลจะอัปเดตเข้าสู่แดชบอร์ดทันที

1 ข้อมูลวันเวลา แผนก และเครื่องจักร

2 รายละเอียดใบสั่งผลิต ปัญหาที่พบ และจำนวนของเสีย

3 ผู้รับผิดชอบและรายละเอียดการแก้ไข

ฐานข้อมูลประวัติการแจ้งปัญหาทั้งหมด

ค้นหา แก้ไขสถานะ ส่งออกไฟล์ Excel (CSV) หรือเชื่อมต่อ Google Sheets

วันที่ / กะ แผนก / เครื่อง Job No. / ลูกค้า-ชื่องาน ปัญหาที่พบ ของเสีย ผู้ผลิตงาน ผู้แจ้ง สาเหตุ / การแก้ไข สถานะ จัดการ

คู่มือเชื่อมต่อฐานข้อมูล Google Sheets (Apps Script)

หากต้องการให้ข้อมูลที่บันทึกจากหน้าเว็บนี้ส่งตรงเข้า Google Sheets ของบริษัทแบบอัตโนมัติ สามารถทำตามขั้นตอนนี้ได้เลยครับ:

  1. สร้างไฟล์ Google Sheets ใหม่ ตั้งชื่อชีตว่า DefectLogs
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. คัดลอกโค้ดด้านล่างไปวาง แล้วกด ทำให้ใช้งานได้ (Deploy) > รายการใหม่ (New deployment) > เว็บแอป (Web app) (ตั้งค่าสิทธิ์เป็น ทุกคน / Anyone)
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DefectLogs");
  if (!sheet) {
    sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("DefectLogs");
    sheet.appendRow(["วันที่", "กะ", "แผนก", "เครื่องจักร", "Job No.", "ลูกค้า/ชื่องาน", "ปัญหาที่พบ", "จำนวน", "หน่วย", "ผู้ผลิตงาน", "ผู้แจ้งข้อมูล", "สาเหตุ/การแก้ไข", "สถานะ"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.date, data.shift, data.dept, data.machine, data.jobNo,
    data.customerJob, data.defect, data.qty, data.unit,
    data.operator, data.reporter, data.causeAction, data.status
  ]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
